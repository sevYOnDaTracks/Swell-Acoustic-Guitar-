cmake_minimum_required(VERSION 3.22)
project(SwellAcousticGuitar VERSION 0.3.0 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(GUITAR_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND GUITAR_FORMATS AU)
endif()
# Never change PLUGIN_CODE / BUNDLE_ID once released: saved DAW projects would lose the plugin.
juce_add_plugin(SwellAcousticGuitar COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.acoustic-guitar" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Sagt FORMATS ${GUITAR_FORMATS} PRODUCT_NAME "Swell Guitar" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
file(GLOB GUITAR_SAMPLES CONFIGURE_DEPENDS assets/samples/*/*.flac)
juce_add_binary_data(GuitarAssets SOURCES assets/guitar-photo.jpg assets/guitar-photo-electric.jpg ${GUITAR_SAMPLES})
target_sources(SwellAcousticGuitar PRIVATE Source/SampleLibrary.cpp Source/GuitarEngine.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SwellAcousticGuitar PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_compile_options(SwellAcousticGuitar PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_link_libraries(SwellAcousticGuitar PRIVATE GuitarAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(GuitarTest PRODUCT_NAME "Guitar Audio Tests")
target_sources(GuitarTest PRIVATE Tests/AudioTest.cpp)
target_include_directories(GuitarTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(GuitarTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(GuitarTest PRIVATE SwellAcousticGuitar)
add_test(NAME GuitarAudio COMMAND GuitarTest)
juce_add_console_app(GuitarSnapshot PRODUCT_NAME "Guitar Snapshot")
target_sources(GuitarSnapshot PRIVATE Tools/Snapshot.cpp)
target_include_directories(GuitarSnapshot PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(GuitarSnapshot PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(GuitarSnapshot PRIVATE SwellAcousticGuitar)
